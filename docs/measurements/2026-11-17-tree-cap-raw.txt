## meas.sh (the harness)
#!/bin/sh
# meas.sh: run the tree-cap measurement; wall time and peak RSS (VmHWM)
# of cmc, sampled every 10 s, written to run.txt
set -u
cd "$(dirname "$0")"
{
	echo "start $(date -u +%FT%TZ) $(cut -d' ' -f1-3 /proc/loadavg)"
	t0=$(date +%s)
	./cmc --model-file spin.model > cmc-out.txt 2>&1 &
	pid=$!
	hwm=0
	while kill -0 "$pid" 2>/dev/null; do
		h=$(awk '/^VmHWM:/ {print $2}' "/proc/$pid/status" 2>/dev/null)
		[ -n "$h" ] && hwm=$h
		echo "$(($(date +%s) - t0)) s VmHWM ${hwm} kB load $(cut -d' ' -f1-3 /proc/loadavg)" > progress.txt
		sleep 10
	done
	wait "$pid"
	rc=$?
	t1=$(date +%s)
	echo "end $(date -u +%FT%TZ) $(cut -d' ' -f1-3 /proc/loadavg)"
	echo "exit $rc wall $((t1 - t0)) s peak VmHWM ${hwm} kB (last 10 s sample)"
} > run.txt 2>&1
